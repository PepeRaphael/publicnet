import json import re import warnings import networkx as nx import numpy as np from sklearn.cluster import SpectralClustering import yaml from pyscript import window, document import js # Suppression des alertes WebAssembly warnings.filterwarnings("ignore", category=RuntimeWarning, module="threadpoolctl") def snap(val): """ Fonction de capture algébrique. Force les coordonnées sur une grille mathématique stricte de 5 unités. """ return int(round(float(val) / 5.0) * 5.0) def parse_true_topology(raw_data): """Extraction spatiale résolvant l'illusion des labels et des flottants.""" data = json.loads(raw_data) shape_data = data.get("schematics", [{}])[0].get("dataStr", {}).get("shape", []) spatial_graph = nx.Graph() components = {} net_labels = {} for item in shape_data: if not isinstance(item, str): continue # 1. Traitement des fils physiques if item.startswith('W~'): parts = item.split('~') coords = parts[1].split(' ') for i in range(0, len(coords)-2, 2): p1 = (snap(coords[i]), snap(coords[i+1])) p2 = (snap(coords[i+2]), snap(coords[i+3])) spatial_graph.add_edge(p1, p2) # 2. Traitement des Nœuds Logiques (NetLabels) elif item.startswith('N~'): parts = item.split('~') if len(parts) >= 6: px, py = snap(parts[1]), snap(parts[2]) net_name = parts[5] # Le nom du réseau (GND, 5v, etc.) if net_name not in net_labels: net_labels[net_name] = [] net_labels[net_name].append((px, py)) spatial_graph.add_node((px, py)) # 3. Traitement des Composants et de leurs Broches elif item.startswith('LIB~'): id_match = re.search(r'comment~([A-Z0-9_]+)~1~start', item) type_match = re.search(r'package`([^`]+)`', item) alias_match = re.search(r'comment~([^~]+)~1~start.*comment~([A-Z0-9_]+)~1~start', item) if id_match: comp_id = id_match.group(1) comp_type = type_match.group(1) if type_match else "UNK" if alias_match and alias_match.group(1) != comp_id: comp_type = f"{comp_type} ({alias_match.group(1)})" components[comp_id] = {'type': comp_type, 'pins': {}} # Découpage des blocs de broches for segment in item.split('#@$P~')[1:]: parts = segment.split('~') if len(parts) > 3: pin_num = parts[2] # Index corrigé pour le numéro de broche coord_match = re.search(r'\^\^([-\d.]+)~([-\d.]+)\^\^', segment) if coord_match: px = snap(coord_match.group(1)) py = snap(coord_match.group(2)) components[comp_id]['pins'][pin_num] = (px, py) spatial_graph.add_node((px, py)) # 4. Fusion Algorithmique # Création des arêtes virtuelles entre les labels de même nom for name, coords in net_labels.items(): for i in range(len(coords)-1): spatial_graph.add_edge(coords[i], coords[i+1]) # Recherche des composantes connexes (Équipotentielles) nets = {} for idx, cc in enumerate(nx.connected_components(spatial_graph)): net_name = f"NET_AUTO_{idx}" # Écrasement du nom si un label logique fait partie de ce sous-graphe for node in cc: for name, coords in net_labels.items(): if node in coords: net_name = name break # Collecte des broches incluses dans cette équipotentielle connected_pins = [] for cid, cdata in components.items(): for pnum, pcoord in cdata['pins'].items(): if pcoord in cc: connected_pins.append(f"{cid}.{pnum}") if len(connected_pins) > 1: nets[net_name] = connected_pins return components, nets def estimate_optimal_k(adj_matrix): """Calcul spectral pour trouver le saut optimal de l'Eigengap.""" n = adj_matrix.shape[0] if n < 3: return 1 d = np.sum(adj_matrix, axis=1) d_inv_sqrt = np.power(d, -0.5, where=d!=0) D_inv_sqrt = np.diag(d_inv_sqrt) L_sym = np.eye(n) - np.dot(D_inv_sqrt, np.dot(adj_matrix, D_inv_sqrt)) eigenvalues = np.linalg.eigvalsh(L_sym) gaps = np.diff(eigenvalues) max_k = max(2, min(n // 3, 10)) optimal_k = np.argmax(gaps[:max_k]) + 1 return max(2, optimal_k) def generate_rich_llm_netlist(raw_json): # Parsing géométrique strict components, nets = parse_true_topology(raw_json) comp_keys = list(components.keys()) n_comps = len(comp_keys) # Matrice de projection adj_matrix = np.zeros((n_comps, n_comps)) for net_name, pins in nets.items(): deg = len(pins) if deg > 1: # Pénalisation inverse pour isoler les réseaux d'alimentation (VCC/GND) weight = 1.0 / np.log(deg + 1) cids = list(set([p.split('.')[0] for p in pins if p.split('.')[0] in comp_keys])) for i in range(len(cids)): for j in range(i+1, len(cids)): idx1 = comp_keys.index(cids[i]) idx2 = comp_keys.index(cids[j]) adj_matrix[idx1, idx2] += weight adj_matrix[idx2, idx1] += weight # Clustering Spectral k = estimate_optimal_k(adj_matrix) sc = SpectralClustering(n_clusters=k, affinity='precomputed', assign_labels='kmeans') labels = sc.fit_predict(adj_matrix) topology = {f"BLOCK_{i}": [] for i in range(k)} for cid, label in zip(comp_keys, labels): topology[f"BLOCK_{label}"].append({ "id": cid, "type": components[cid]['type'] }) payload = { "system_instruction": "Agis en tant qu'ingénieur expert. Diagnostique cette architecture dont la topologie a été résolue par analyse spectrale.", "metrics": { "components_detected": n_comps, "nets_resolved": len(nets), "functional_blocks_found": int(k) }, "topology": topology, "connectivity_matrix": nets } return yaml.dump(payload, allow_unicode=True, default_flow_style=False, sort_keys=False) def run_engine(event): try: js.updateStatus("Extraction rigoureuse de la géométrie en cours...") raw_json = window.easyEdaRawData yaml_output = generate_rich_llm_netlist(raw_json) js.triggerDownload(yaml_output) except Exception as e: js.updateStatus(f"Erreur algorithmique : {str(e)}") document.getElementById("loader").style.display = "none" from pyodide.ffi import create_proxy proxy = create_proxy(run_engine) document.addEventListener("trigger_python_engine", proxy)
