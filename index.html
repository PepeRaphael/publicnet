import json
import networkx as nx
import numpy as np
from sklearn.cluster import SpectralClustering
import yaml
from pyscript import window, document
from pyodide.ffi import create_proxy

# Masque l'avertissement de threadpoolctl lié à l'absence de libc dans WebAssembly
warnings.filterwarnings("ignore", category=RuntimeWarning, module="threadpoolctl")

def run_llmnet_engine(event):
    raw_json = window.easyEdaRawData
    if not raw_json:
        return
        
    try:
        data = json.loads(raw_json)
        
        # 1. Simulation de l'extraction des Nets (simplifiée pour l'architecture)
        # Dans le système final, vous intégrerez ici les lois des nœuds de Kirchhoff
        components_list = data.get("shape", [])
        
        # 2. Simulation d'une matrice Laplacienne pour la preuve de concept
        # Génération d'une topologie factice basée sur la taille du fichier
        n_nodes = max(3, min(10, len(raw_json) // 1000))
        adj_matrix = np.ones((n_nodes, n_nodes)) - np.eye(n_nodes)
        
        # 3. Clustering Spectral
        sc = SpectralClustering(n_clusters=2, affinity='precomputed')
        labels = sc.fit_predict(adj_matrix)
        
        # 4. Génération YAML Sémantique
        output = {
            "system_instruction": "Architecture topologique analysée côté client.",
            "metrics": {
                "file_size_bytes": len(raw_json),
                "detected_nodes": n_nodes
            },
            "spectral_clusters": {
                f"BLOCK_{i}": int(label) for i, label in enumerate(labels)
            }
        }
        
        yaml_output = yaml.dump(output, default_flow_style=False)
        
        # 5. Rendu dans le DOM HTML
        output_el = document.getElementById("output")
        output_el.style.display = "block"
        output_el.innerText = yaml_output
        document.getElementById("status-text").innerText = "Analyse terminée avec succès."
        
    except Exception as e:
        document.getElementById("status-text").innerText = f"Erreur d'analyse mathématique : {str(e)}"

# Écoute de l'événement JavaScript
proxy = create_proxy(run_llmnet_engine)
document.addEventListener("trigger_python_engine", proxy)
